## 💾 DATABASE SCHEMA (SQLite)


-- Customers
CREATE TABLE customers (
id INTEGER PRIMARY KEY AUTOINCREMENT,
name TEXT NOT NULL,
email TEXT UNIQUE,
phone TEXT,
language_preference TEXT DEFAULT 'en',
created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);
-- Vehicles
CREATE TABLE vehicles (
id INTEGER PRIMARY KEY AUTOINCREMENT,
customer_id INTEGER,
vin TEXT UNIQUE,
model TEXT,
year INTEGER,
mileage INTEGER,
last_service_date DATE,
FOREIGN KEY (customer_id) REFERENCES customers(id)
);
-- Vehicle Health Logs
CREATE TABLE vehicle_health_logs (
id INTEGER PRIMARY KEY AUTOINCREMENT,
vehicle_id INTEGER,
battery_health INTEGER, -- 0-100%
engine_oil TEXT, -- Good/Low/Critical
brake_pad_thickness REAL, -- mm
tire_pressure_fl REAL, -- PSI
tire_pressure_fr REAL,
tire_pressure_rl REAL,
tire_pressure_rr REAL,
engine_temp REAL, -- Celsius
coolant_level TEXT,
recorded_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (vehicle_id) REFERENCES vehicles(id)
);
-- Conversations
CREATE TABLE conversations (
id INTEGER PRIMARY KEY AUTOINCREMENT,
customer_id INTEGER,
channel TEXT, -- whatsapp/web/app
status TEXT, -- active/resolved/escalated
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
closed_at DATETIME,
FOREIGN KEY (customer_id) REFERENCES customers(id)
);
-- Messages
CREATE TABLE messages (
id INTEGER PRIMARY KEY AUTOINCREMENT,
conversation_id INTEGER,
sender TEXT, -- customer/ai/agent
message TEXT,
sentiment_score REAL, -- 0-1
emotion_label TEXT, -- happy/neutral/concerned/frustrated/urgent
timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (conversation_id) REFERENCES conversations(id)
);
-- Maintenance Alerts
CREATE TABLE maintenance_alerts (
id INTEGER PRIMARY KEY AUTOINCREMENT,
vehicle_id INTEGER,
alert_type TEXT, -- routine/wear_warning/critical
component TEXT, -- brake_pads/oil/battery
message TEXT,
severity TEXT, -- low/medium/high
predicted_failure_date DATE,
status TEXT, -- pending/acknowledged/booked/dismissed
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (vehicle_id) REFERENCES vehicles(id)
);
-- Service Bookings
CREATE TABLE service_bookings (
id INTEGER PRIMARY KEY AUTOINCREMENT,
customer_id INTEGER,
vehicle_id INTEGER,
booking_id TEXT UNIQUE,
service_type TEXT,
service_center TEXT,
scheduled_date DATE,
scheduled_time TIME,
status TEXT, -- confirmed/completed/cancelled
estimated_cost_min INTEGER,
estimated_cost_max INTEGER,
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (customer_id) REFERENCES customers(id),
FOREIGN KEY (vehicle_id) REFERENCES vehicles(id)
);
-- Feedback
CREATE TABLE feedback (
id INTEGER PRIMARY KEY AUTOINCREMENT,
conversation_id INTEGER,
rating INTEGER, -- 1-5 stars
was_helpful BOOLEAN,
comment TEXT,
created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (conversation_id) REFERENCES conversations(id)
);
-- Knowledge Base
CREATE TABLE knowledge_base (
id INTEGER PRIMARY KEY AUTOINCREMENT,
category TEXT,
question TEXT,
answer TEXT,
keywords TEXT, -- comma-separated for search
source TEXT -- manual page reference
);
